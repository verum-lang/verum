use verum_vbc::archive::read_archive_from_file;
#[test]
fn report_exact_archive_return_descriptors() {
 let path = std::env::var("VERUM_T1698_ARCHIVE").expect("explicit diagnostic input");
 let archive=read_archive_from_file(path).expect("archive");
 eprintln!("archive modules={:?}", archive.module_names_ordered());
 for owner in archive.module_names_ordered() {
  let module=archive.load_module(owner).expect(owner);
  for f in &module.functions {
   let name=module.get_string(f.name).unwrap_or("?");
   if ["UInt64.to_be_bytes", "UInt32.to_be_bytes", "Sha256.finalize"].iter().any(|n|name.ends_with(n)) {
    eprintln!("{owner}::{name} id={:?} ret={:?} params={:?}",f.id,f.return_type,f.params);
   }
  }
 }
}
